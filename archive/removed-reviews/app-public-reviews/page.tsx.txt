import type { Metadata } from "next";
import Link from "next/link";
import { ctaClass } from "@/components/CtaButton";
import { Note, PageHeader, Section } from "@/components/primitives";
import { ReviewCard } from "@/components/ReviewCard";
import { ReviewFigureRow } from "@/components/ReviewFigureRow";
import { StatTiles } from "@/components/StatTiles";
import { getEditorialReview } from "@/lib/data/editorial";
import { formatInt, formatRating } from "@/lib/format";
import { getRegistry, getSiteStats, loadSectors } from "@/lib/queries";
import { mostReviewed, reviewsBySector } from "@/lib/review-summary";

export const revalidate = 3600;

export const metadata: Metadata = {
  title: "Reviews: pros, cons and a verdict for every product",
  description:
    "A written review of every product on the register: what it does well, where it falls short, who it suits and what it costs in rand, with its rating and the reviews analysed beside it.",
  alternates: { canonical: "/reviews" },
};

export default async function ReviewsHubPage() {
  const [stats, reg, sectors] = await Promise.all([getSiteStats(), getRegistry(), loadSectors()]);

  const bySector = reviewsBySector(reg.ranked, sectors);
  const top = mostReviewed(reg.ranked, 6);
  const maxReviews = top[0]?.review_count ?? 1;
  const registerMean = reg.priors.meanRating;

  return (
    <>
      <PageHeader title="Reviews" stamp={`${stats.products} products reviewed`}>
        <p>
          A written review of every product on the register: what it does well, where it falls short, who it suits and what it costs, with its rating and the reviews analysed beside it. Each review is read from the vendor&apos;s own pages and carries the date it was read.
        </p>
      </PageHeader>

      <section aria-label="The review figures" className="mx-auto w-full max-w-[1080px] px-6 pb-8">
        <StatTiles
          tiles={[
            { value: String(stats.products), label: "Products", note: "each with a written review", href: "/register" },
            { value: formatInt(stats.reviews), label: "Reviews analysed", note: "never called verified", href: "/method" },
            { value: String(stats.sectors), label: "Categories", note: "accounting to projects", href: "/sectors" },
            { value: String(stats.comparisons), label: "Comparisons", note: "two products at a time", href: "/versus" },
          ]}
        />
        <Note className="mt-4 max-w-[760px]">
          The register mean is {formatRating(registerMean)} out of 5, weighted by review count. Read how the figures were produced in the{" "}
          <Link href="/method" className="prose-link">method</Link>.
        </Note>

        <nav aria-label="Jump to a category" className="mt-8 flex flex-wrap gap-2">
          {bySector.map(({ sector, items }) => (
            <a
              key={sector.slug}
              href={`#sector-${sector.slug}`}
              className="inline-flex min-h-10 items-center gap-2 rounded-full border border-line-firm bg-surface px-4 text-[14px] font-semibold hover:bg-lime-wash"
            >
              {sector.name}
              <span className="stamp text-muted">{items.length}</span>
            </a>
          ))}
        </nav>
      </section>

      {bySector.map(({ sector, items, reviews, mean }, i) => (
        <section key={sector.slug} id={`sector-${sector.slug}`} aria-labelledby={`sec-${sector.slug}`} className="mx-auto w-full max-w-[1080px] scroll-mt-24 px-6">
          <div className={`${i === 0 ? "" : "border-t border-line-firm"} py-10 lg:py-14`}>
            <header className="mb-6 flex flex-wrap items-end justify-between gap-x-8 gap-y-3">
              <div>
                <h2 id={`sec-${sector.slug}`} className="font-heading text-[28px] leading-8 font-bold tracking-[-0.02em]">
                  <Link href={`/sector/${sector.slug}`} className="hover:underline hover:underline-offset-4">
                    {sector.label}
                  </Link>
                </h2>
                <p className="mt-1 max-w-[56ch] text-[14px] leading-5 text-muted">{sector.description}</p>
              </div>
              <dl className="flex gap-8">
                <div>
                  <dt className="stamp text-muted">Products</dt>
                  <dd className="m-0 font-heading text-[24px] leading-7 font-bold tracking-[-0.02em]">{items.length}</dd>
                </div>
                <div>
                  <dt className="stamp text-muted">Reviews analysed</dt>
                  <dd className="m-0 font-heading text-[24px] leading-7 font-bold tracking-[-0.02em]">{formatInt(reviews)}</dd>
                </div>
                <div>
                  <dt className="stamp text-muted">Mean rating</dt>
                  <dd className="m-0 font-heading text-[24px] leading-7 font-bold tracking-[-0.02em]">{formatRating(mean)}</dd>
                </div>
              </dl>
            </header>
            <ul className="grid gap-5 lg:grid-cols-2">
              {[...items]
                .sort((a, b) => a.rank - b.rank)
                .map((s) => (
                  <li key={s.slug}>
                    <ReviewCard s={s} r={getEditorialReview(s.slug)} />
                  </li>
                ))}
            </ul>
          </div>
        </section>
      ))}

      <Section title="Where the review weight sits" aside={<span className="stamp text-muted">Top {top.length} by review count</span>}>
        <ul className="divide-y divide-line rounded-[16px] border border-line-firm bg-surface px-5">
          {top.map((s, i) => (
            <ReviewFigureRow key={s.slug} s={s} maxReviews={maxReviews} position={i + 1} />
          ))}
        </ul>
      </Section>

      <Section title="How to read these reviews">
        <ol className="grid gap-5 md:grid-cols-3">
          {[
            {
              n: "01",
              title: "Written from the vendor's own pages",
              body: "Each review records what the vendor publishes, the date we read it and the pages we used. Where a vendor publishes no price or no page, the review says so and makes no claim.",
            },
            {
              n: "02",
              title: "Ratings are analysed, never verified",
              body: "The rating and the count come from a corpus produced by a seeded script, not from verified buyers. That is why we say reviews analysed, and why a written review carries no score of its own.",
            },
            {
              n: "03",
              title: "Yours are different",
              body: "A review submitted here is read by an editor, published as written and kept apart from the corpus. As they arrive they replace the corpus as the evidence behind a rating.",
            },
          ].map((step) => (
            <li key={step.n} className="rounded-[16px] border border-line-firm bg-surface p-6">
              <p className="stamp text-muted">{step.n}</p>
              <h3 className="mt-3 font-heading text-[20px] leading-6 font-bold tracking-[-0.01em]">{step.title}</h3>
              <p className="mt-3 text-[14px] leading-6 text-muted">{step.body}</p>
            </li>
          ))}
        </ol>
        <div className="mt-6 flex flex-wrap gap-3">
          <Link href="/method" className={ctaClass("quiet", "min-h-12 px-7")}>
            How we rank
          </Link>
          <Link href="/register" className={ctaClass("primary", "min-h-12 px-7")}>
            Browse all software <span aria-hidden="true">→</span>
          </Link>
        </div>
      </Section>
    </>
  );
}
