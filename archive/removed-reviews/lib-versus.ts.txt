import { BANKS, LEGAL_ITEMS, SARS_ITEMS, SECTOR_COMPLIANCE } from "./data/sectors";
import { basisLabel, formatDateStamp, formatInt, formatRating, priceLabel } from "./format";
import { matrixCell, sarsStatus, type MatrixCell } from "./registry";
import type { RankedSoftware } from "./types";

export type VCell = { type: "text"; text: string; muted?: boolean } | { type: "mark"; cell: MatrixCell };

export interface VRow {
  label: string;
  a: VCell;
  b: VCell;
  /** Which side wins this row, if either does. */
  win: "a" | "b" | null;
}

export interface VGroup {
  title: string;
  rows: VRow[];
}

const text = (t: string, muted = false): VCell => ({ type: "text", text: t, muted });
const mark = (cell: MatrixCell): VCell => ({ type: "mark", cell });

function higher(a: number, b: number): "a" | "b" | null {
  return a === b ? null : a > b ? "a" : "b";
}

const MARK_RANK: Record<MatrixCell, number> = { yes: 2, no: 1, unchecked: 0, na: 0 };

function markWin(a: MatrixCell, b: MatrixCell): "a" | "b" | null {
  return MARK_RANK[a] === MARK_RANK[b] ? null : MARK_RANK[a] > MARK_RANK[b] ? "a" : "b";
}

/** Lower starting price wins, compared in rand, but only when both are on the same VAT footing. */
function cheaper(a: RankedSoftware, b: RankedSoftware): "a" | "b" | null {
  if (a.starting_price === null || b.starting_price === null) return null;
  if (a.price_basis !== b.price_basis && (a.price_basis?.includes("vat") || b.price_basis?.includes("vat"))) return null;
  const ra = a.price_rand ?? a.starting_price;
  const rb = b.price_rand ?? b.starting_price;
  return ra === rb ? null : ra < rb ? "a" : "b";
}

function checklist(
  title: string,
  items: string[],
  a: RankedSoftware,
  b: RankedSoftware,
  read: (s: RankedSoftware) => string[] | null,
  summaryLabel: string,
): VGroup {
  const aData = read(a);
  const bData = read(b);
  const summary = (data: string[] | null): VCell =>
    data === null ? mark("unchecked") : text(data.length ? `${data.length} recorded` : "None recorded");
  const rows: VRow[] = [
    {
      label: summaryLabel,
      a: summary(aData),
      b: summary(bData),
      win: aData && bData ? higher(aData.length, bData.length) : null,
    },
  ];
  // Item level rows only appear once at least one side has been checked. Six rows of Unchecked say nothing.
  if (aData !== null || bData !== null) {
    for (const item of items) {
      const ca: MatrixCell = aData === null ? "unchecked" : aData.includes(item) ? "yes" : "no";
      const cb: MatrixCell = bData === null ? "unchecked" : bData.includes(item) ? "yes" : "no";
      rows.push({ label: item, a: mark(ca), b: mark(cb), win: markWin(ca, cb) });
    }
  }
  return { title, rows };
}

/**
 * The comparison matrix. Rows are grouped into Pricing, SARS filing, Bank
 * feeds, Compliance, Features, Support and Availability. The winning cell in a
 * row renders in --ink 600 and the loser in --muted 400, so a row can be read
 * without comparing numbers by eye.
 */
export function buildMatrix(a: RankedSoftware, b: RankedSoftware): VGroup[] {
  const pa = priceLabel(a);
  const pb = priceLabel(b);

  const pricing: VGroup = {
    title: "Pricing",
    rows: [
      {
        label: "Starting price",
        a: text(pa.original ? pa.amount + " (" + pa.original + ")" : pa.amount, pa.kind === "quoted"),
        b: text(pb.original ? pb.amount + " (" + pb.original + ")" : pb.amount, pb.kind === "quoted"),
        win: cheaper(a, b),
      },
      { label: "Basis", a: text(basisLabel(a.price_basis) || "Not applicable", !a.price_basis), b: text(basisLabel(b.price_basis) || "Not applicable", !b.price_basis), win: null },
      {
        label: "Billed by the vendor in",
        a: text(a.price_currency ?? "Not published", !a.price_currency),
        b: text(b.price_currency ?? "Not published", !b.price_currency),
        // Rand billing beats foreign billing for a rand earner, because it carries no exchange rate risk.
        win: a.price_currency && b.price_currency && a.price_currency !== b.price_currency ? (a.price_currency === "ZAR" ? "a" : b.price_currency === "ZAR" ? "b" : null) : null,
      },
      {
        label: "Price checked",
        a: a.price_checked_at ? text(formatDateStamp(a.price_checked_at)) : mark("unchecked"),
        b: b.price_checked_at ? text(formatDateStamp(b.price_checked_at)) : mark("unchecked"),
        win: a.price_checked_at && b.price_checked_at ? (a.price_checked_at === b.price_checked_at ? null : a.price_checked_at > b.price_checked_at ? "a" : "b") : a.price_checked_at ? "a" : b.price_checked_at ? "b" : null,
      },
    ],
  };

  const applicable = new Set([...(SECTOR_COMPLIANCE[a.sector] ?? []), ...(SECTOR_COMPLIANCE[b.sector] ?? [])]);
  const sarsItems = SARS_ITEMS.filter((i) => applicable.has(i));
  const legalItems = LEGAL_ITEMS.filter((i) => applicable.has(i));

  const sars = checklist("SARS filing", sarsItems, a, b, (s) => (s.compliance === null ? null : s.compliance.filter((c) => (SARS_ITEMS as readonly string[]).includes(c) || c === "eFiling")), "SARS items recorded");
  const transfer = (s: RankedSoftware): VCell => {
    const st = sarsStatus(s);
    return st === "unchecked" ? mark("unchecked") : text(st === "files" ? "Files" : "Manual");
  };
  sars.rows.unshift({
    label: "Transfers into SARS",
    a: transfer(a),
    b: transfer(b),
    win: sarsStatus(a) === sarsStatus(b) ? null : sarsStatus(a) === "files" ? "a" : sarsStatus(b) === "files" ? "b" : null,
  });

  const banks = checklist("Bank feeds", [...BANKS], a, b, (s) => s.bank_feeds, "Banks recorded");
  const compliance = checklist("Compliance", legalItems, a, b, (s) => (s.compliance === null ? null : s.compliance.filter((c) => (LEGAL_ITEMS as readonly string[]).includes(c))), "Items recorded");

  const featureUnion = [...new Set([...a.features, ...b.features])];
  const features: VGroup = {
    title: "Features",
    rows: featureUnion.length
      ? featureUnion.map((f) => {
          const ca: MatrixCell = a.features.includes(f) ? "yes" : "no";
          const cb: MatrixCell = b.features.includes(f) ? "yes" : "no";
          return { label: f, a: mark(ca), b: mark(cb), win: markWin(ca, cb) };
        })
      : [{ label: "Features recorded", a: mark("unchecked"), b: mark("unchecked"), win: null }],
  };

  const support: VGroup = {
    title: "Support",
    rows: [
      {
        label: "Support rating",
        a: a.subratings ? text(formatRating(a.subratings.support)) : mark("unchecked"),
        b: b.subratings ? text(formatRating(b.subratings.support)) : mark("unchecked"),
        win: a.subratings && b.subratings ? higher(a.subratings.support, b.subratings.support) : null,
      },
      {
        label: "Reviews analysed",
        a: text(formatInt(a.review_count)),
        b: text(formatInt(b.review_count)),
        win: higher(a.review_count, b.review_count),
      },
    ],
  };

  const flag = (v: boolean | null): MatrixCell => (v === null ? "unchecked" : v ? "yes" : "no");
  const availability: VGroup = {
    title: "Availability",
    rows: [
      { label: "Free trial", a: mark(flag(a.free_trial)), b: mark(flag(b.free_trial)), win: markWin(flag(a.free_trial), flag(b.free_trial)) },
      { label: "Free plan", a: mark(flag(a.free_plan)), b: mark(flag(b.free_plan)), win: markWin(flag(a.free_plan), flag(b.free_plan)) },
      { label: "Demo", a: mark(flag(a.demo)), b: mark(flag(b.demo)), win: markWin(flag(a.demo), flag(b.demo)) },
    ],
  };

  return [pricing, sars, banks, compliance, features, support, availability];
}

/** Which side holds the winning figure in the six numeral band. */
export function figureWins(a: RankedSoftware, b: RankedSoftware) {
  return {
    rating: higher(a.rating, b.rating),
    reviews: higher(a.review_count, b.review_count),
    price: cheaper(a, b),
  };
}

export { matrixCell };
