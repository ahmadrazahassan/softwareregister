/**
 * The register.
 *
 * What is recorded here comes from the build specification: names, sectors,
 * list prices, price bases, billing currency, the two real check dates, and
 * the rating and review count behind each product.
 *
 * What is NOT recorded is left null on purpose, and the site renders null as
 * "Unchecked". That covers `compliance` (SARS filing, UIF, BCEA and so on),
 * `bank_feeds`, `free_trial`, `demo`, `subratings`, features, integrations and
 * screenshots. A trust register must not guess at whether a product files
 * VAT201 into eFiling. Fill those in through the admin Products screen or by
 * editing this file, and every filter, matrix and fact strip lights up.
 *
 * `price_currency` is recorded for any product with a published price, and for
 * the two quote only products whose vendors bill in a fixed foreign currency
 * (BambooHR in USD, Personio in EUR). It is null for the other quote only
 * products, because a quote leaves the currency unpublished.
 */

  basis?: PriceBasis;
  checked?: string;
  sage?: boolean;
}

function toSoftware(r: Row, i: number): Software {
  const free = r.price === 0;
  const quoted = r.price === undefined;
  return {
    id: `sw_${String(i + 1).padStart(2, "0")}`,
    slug: r.slug,
    name: r.name,
    vendor: r.vendor,
    sector: r.sector,
    summary: r.summary,
    website_url: r.site,
    vendor_website: r.host,
    affiliate_url: r.sage ? SAGE_AFFILIATE_URL : null,
    founded: r.founded ?? null,
    rating: r.rating,
    review_count: r.reviews,
    starting_price: quoted ? null : r.price!,
    price_currency: r.currency ?? null,
    price_basis: r.basis ?? null,
    price_checked_at: r.checked ?? null,
    pricing_plans: [
      {
        name: free ? "Free tier" : quoted ? "Quoted plan" : "Starting plan",
        price: quoted ? null : r.price!,
        currency: r.currency ?? null,
        basis: r.basis ?? null,
        adds: null,
      },
    ],
    pricing_note: null,
    free_trial: null,
    free_plan: free ? true : null,
    demo: null,
    compliance: null,
    bank_feeds: null,
    features: [],
    integrations: [],
    subratings: null,
    assessment: null,
    logo: LOGOS[r.slug]?.src ?? null,
    screenshots: (SCREENSHOTS[r.slug] ?? []).map((shot) => shot.src),
    status: "active",
  };
}

export const SOFTWARE: Software[] = ROWS.map(toSoftware);
