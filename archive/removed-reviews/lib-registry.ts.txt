/**
 * Pure functions over the dataset. Nothing in here touches the network, the
 * filesystem or React, so the same code runs in pages, in the seed script and
 * in scripts/verify-dataset.ts.
 */

export interface Registry {
  ranked: RankedSoftware[];
  priors: PriorValues;
  bySlug: Map<string, RankedSoftware>;
}

export function buildRegistry(software: Software[], rates: FxRates = FX_SNAPSHOT): Registry {
  const active = software.filter((s) => s.status === "active");
  const { ranked: scored, priors } = rankSoftware(active);
  // Every price also gets a rand figure. A foreign price is converted at the reference rate, and the rate travels with it.
  const ranked: RankedSoftware[] = scored.map((s) => {
    const fx = fxFor(s.price_currency, rates);
    const price_rand = s.starting_price === null ? null : fx ? s.starting_price * fx.rand_per_unit : s.starting_price;
    return { ...s, price_rand, fx };
  });
  return { ranked, priors, bySlug: new Map(ranked.map((s) => [s.slug, s])) };
}

export function sectorOf(slug: SectorSlug | string, sectors: Sector[] = SECTORS): Sector | undefined {
  return sectors.find((s) => s.slug === slug);
}

export const CURRENCY_NAME: Record<Currency, string> = {
  ZAR: "rand",
  USD: "US dollars",
  GBP: "pounds sterling",
  EUR: "euros",
};

export const isForeign = (c: Currency | null): boolean => c === "USD" || c === "GBP" || c === "EUR";

/* ------------------------------------------------------------------ stats */

export function computeStats(
  reg: Registry,
  extra: { sectors: number; comparisons: number; briefings: number },
): SiteStats {
  const items = reg.ranked;
  const dates = items.map((s) => s.price_checked_at).filter((d): d is string => Boolean(d));
  return {
    products: items.length,
    sectors: extra.sectors,
    comparisons: extra.comparisons,
    briefings: extra.briefings,
    reviews: items.reduce((sum, s) => sum + s.review_count, 0),
    priced: items.filter((s) => (s.starting_price ?? 0) > 0).length,
    free: items.filter((s) => s.starting_price === 0).length,
    quoted: items.filter((s) => s.starting_price === null).length,
    zar: items.filter((s) => s.price_currency === "ZAR").length,
    foreign: items.filter((s) => isForeign(s.price_currency)).length,
    unrecordedCurrency: items.filter((s) => s.price_currency === null).length,
    lastChecked: dates.length ? dates.reduce((a, b) => (a > b ? a : b)) : null,
    checkedCount: dates.length,
  };
}

/* --------------------------------------------------------------- SARS etc */

export type SarsStatus = "unchecked" | "files" | "manual";

/** Files means the product transfers into eFiling or e@syFile. Manual means it was checked and does not. */
export function sarsStatus(s: Software): SarsStatus {
  if (s.compliance === null) return "unchecked";
  return s.compliance.some((c) => (SARS_TRANSFER_ITEMS as readonly string[]).includes(c)) ? "files" : "manual";
}

/* ----------------------------------------------------------- register query */

export type SortKey = "rank" | "name" | "sector" | "rating" | "reviews" | "price" | "checked";
export type PriceBand = "free" | "under300" | "300to1000" | "quoted";
export type Availability = "trial" | "plan" | "demo";

export interface RegisterQuery {
  sector: string[];
  price: PriceBand | null;
  currency: Currency[];
  sars: string[];
  legal: string[];
  bank: string[];
  avail: Availability[];
  sort: SortKey;
  dir: "asc" | "desc";
  /** Cards are the default. The table is for people who want every column at once. */
  view: "cards" | "table";
}

type Params = Record<string, string | string[] | undefined>;

function many(v: string | string[] | undefined): string[] {
  if (v === undefined) return [];
  return (Array.isArray(v) ? v : [v]).flatMap((x) => x.split(",")).map((x) => x.trim()).filter(Boolean);
}

function one(v: string | string[] | undefined): string | null {
  if (v === undefined) return null;
  return (Array.isArray(v) ? v[0] : v) ?? null;
}

const SORT_KEYS: SortKey[] = ["rank", "name", "sector", "rating", "reviews", "price", "checked"];
const PRICE_BANDS: PriceBand[] = ["free", "under300", "300to1000", "quoted"];
const CURRENCIES: Currency[] = ["ZAR", "USD", "GBP", "EUR"];
const AVAIL: Availability[] = ["trial", "plan", "demo"];

export function parseRegisterQuery(sp: Params): RegisterQuery {
  const sort = one(sp.sort) as SortKey | null;
  const price = one(sp.price) as PriceBand | null;
  const dir = one(sp.dir);
  const sortKey: SortKey = sort && SORT_KEYS.includes(sort) ? sort : "rank";
  const defaultDir: "asc" | "desc" = ["rating", "reviews", "checked"].includes(sortKey) ? "desc" : "asc";
  return {
    sector: many(sp.sector),
    price: price && PRICE_BANDS.includes(price) ? price : null,
    currency: many(sp.currency).filter((c): c is Currency => CURRENCIES.includes(c as Currency)),
    sars: many(sp.sars),
    legal: many(sp.legal),
    bank: many(sp.bank),
    avail: many(sp.avail).filter((a): a is Availability => AVAIL.includes(a as Availability)),
    sort: sortKey,
    dir: dir === "asc" || dir === "desc" ? dir : defaultDir,
    view: one(sp.view) === "table" ? "table" : "cards",
  };
}

export function isFiltered(q: RegisterQuery): boolean {
  return (
    q.sector.length > 0 ||
    q.price !== null ||
    q.currency.length > 0 ||
    q.sars.length > 0 ||
    q.legal.length > 0 ||
    q.bank.length > 0 ||
    q.avail.length > 0
  );
}

/** A /register URL for a query, with optional overrides. Keeps a filtered view shareable. */
export function registerHref(q: RegisterQuery, patch: Partial<RegisterQuery> = {}): string {
  const next = { ...q, ...patch };
  const params = new URLSearchParams();
  for (const v of next.sector) params.append("sector", v);
  if (next.price) params.set("price", next.price);
  for (const v of next.currency) params.append("currency", v);
  for (const v of next.sars) params.append("sars", v);
  for (const v of next.legal) params.append("legal", v);
  for (const v of next.bank) params.append("bank", v);
  for (const v of next.avail) params.append("avail", v);
  if (next.sort !== "rank") {
    params.set("sort", next.sort);
    params.set("dir", next.dir);
  } else if (next.dir === "desc") {
    params.set("dir", "desc");
  }
  if (next.view === "table") params.set("view", "table");
  const qs = params.toString();
  return qs ? `/register?${qs}` : "/register";
}

function matchesPrice(s: RankedSoftware, band: PriceBand): boolean {
  const p = s.starting_price;
  // Bands are rand bands. A foreign price is placed in one by its converted figure.
  const rand = s.price_rand;
  switch (band) {
    case "free":
      return p === 0;
    case "quoted":
      return p === null;
    case "under300":
      return rand !== null && rand > 0 && rand < 300;
    case "300to1000":
      return rand !== null && rand >= 300 && rand <= 1000;
  }
}

/** Sector and currency are OR within the group. SARS, legal, bank and availability must all match. */
export function filterSoftware(items: RankedSoftware[], q: RegisterQuery): RankedSoftware[] {
  return items.filter((s) => {
    if (q.sector.length && !q.sector.includes(s.sector)) return false;
    if (q.price && !matchesPrice(s, q.price)) return false;
    if (q.currency.length && !(s.price_currency && q.currency.includes(s.price_currency))) return false;
    if (q.sars.length && !q.sars.every((c) => s.compliance?.includes(c))) return false;
    if (q.legal.length && !q.legal.every((c) => s.compliance?.includes(c))) return false;
    if (q.bank.length && !q.bank.every((b) => s.bank_feeds?.includes(b))) return false;
    for (const a of q.avail) {
      if (a === "trial" && s.free_trial !== true) return false;
      if (a === "plan" && s.free_plan !== true) return false;
      if (a === "demo" && s.demo !== true) return false;
    }
    return true;
  });
}

/** Free sorts first, then every price by its rand figure, and quoted always sorts last. */
function priceSortKey(s: RankedSoftware): [number, number] {
  if (s.starting_price === null) return [3, 0];
  if (s.starting_price === 0) return [0, 0];
  return [1, s.price_rand ?? s.starting_price];
}

export function sortSoftware(items: RankedSoftware[], q: RegisterQuery): RankedSoftware[] {
  const sign = q.dir === "asc" ? 1 : -1;
  const out = [...items];
  out.sort((a, b) => {
    let cmp = 0;
    switch (q.sort) {
      case "rank":
        cmp = a.rank - b.rank;
        break;
      case "name":
        cmp = a.name.localeCompare(b.name, "en-ZA");
        break;
      case "sector":
        cmp = a.sector.localeCompare(b.sector) || a.rank - b.rank;
        break;
      case "rating":
        cmp = a.rating - b.rating;
        break;
      case "reviews":
        cmp = a.review_count - b.review_count;
        break;
      case "price": {
        const [ga, va] = priceSortKey(a);
        const [gb, vb] = priceSortKey(b);
        // Quoted stays last whichever way the column is sorted.
        if (ga === 3 || gb === 3) return ga === gb ? a.rank - b.rank : ga === 3 ? 1 : -1;
        cmp = ga - gb || va - vb;
        break;
      }
      case "checked": {
        const da = a.price_checked_at;
        const db = b.price_checked_at;
        // Unchecked stays last whichever way the column is sorted.
        if (!da || !db) return da === db ? a.rank - b.rank : !da ? 1 : -1;
        cmp = da.localeCompare(db);
        break;
      }
    }
    return cmp === 0 ? a.rank - b.rank : cmp * sign;
  });
  return out;
}

/* ------------------------------------------------------------------ facets */

export interface Facet {
  value: string;
  label: string;
  count: number;
}

function orderBy(items: string[], canonical: readonly string[]): string[] {
  return [...items].sort((a, b) => {
    const ia = canonical.indexOf(a);
    const ib = canonical.indexOf(b);
    if (ia !== -1 && ib !== -1) return ia - ib;
    if (ia !== -1) return -1;
    if (ib !== -1) return 1;
    return a.localeCompare(b, "en-ZA");
  });
}

function union(items: Software[], pick: (s: Software) => string[] | null): string[] {
  const seen = new Set<string>();
  for (const s of items) for (const v of pick(s) ?? []) seen.add(v);
  return [...seen];
}

export interface Facets {
  sector: Facet[];
  currency: Facet[];
  sars: Facet[];
  legal: Facet[];
  bank: Facet[];
  avail: Facet[];
}

/**
 * Compliance and bank options are the union of values actually present in the
 * data. A hardcoded list would drift from the dataset and offer filters that
 * return nothing.
 */
export function buildFacets(items: RankedSoftware[], sectors: Sector[] = SECTORS): Facets {
  const compliance = union(items, (s) => s.compliance);
  const isSars = (c: string) =>
    (SARS_ITEMS as readonly string[]).includes(c) || (SARS_TRANSFER_ITEMS as readonly string[]).includes(c);
  const sarsValues = orderBy(compliance.filter(isSars), [...SARS_ITEMS, ...SARS_TRANSFER_ITEMS]);
  const legalValues = orderBy(compliance.filter((c) => !isSars(c)), LEGAL_ITEMS);
  const bankValues = orderBy(union(items, (s) => s.bank_feeds), BANKS);

  return {
    sector: sectors.map((sec) => ({
      value: sec.slug,
      label: sec.name,
      count: items.filter((s) => s.sector === sec.slug).length,
    })),
    currency: CURRENCIES.map((c) => ({
      value: c,
      label: c,
      count: items.filter((s) => s.price_currency === c).length,
    })),
    sars: sarsValues.map((v) => ({ value: v, label: v, count: items.filter((s) => s.compliance?.includes(v)).length })),
    legal: legalValues.map((v) => ({ value: v, label: v, count: items.filter((s) => s.compliance?.includes(v)).length })),
    bank: bankValues.map((v) => ({ value: v, label: v, count: items.filter((s) => s.bank_feeds?.includes(v)).length })),
    avail: [
      { value: "trial", label: "Free trial", count: items.filter((s) => s.free_trial === true).length },
      { value: "plan", label: "Free plan", count: items.filter((s) => s.free_plan === true).length },
      { value: "demo", label: "Demo", count: items.filter((s) => s.demo === true).length },
    ],
  };
}

/* --------------------------------------------------------- derived copy */

function billingSentence(s: RankedSoftware): string {
  if (isForeign(s.price_currency)) {
    const was = s.starting_price ? ` The rand figure is converted from ${formatMoney(s.starting_price, s.price_currency)}.` : "";
    return `Billed in ${CURRENCY_NAME[s.price_currency!]}, so its rand cost moves with the exchange rate.${was}`;
  }
  if (s.price_currency === "ZAR") return "Billed in rand.";
  return "The billing currency is not published.";
}

function complianceSentence(s: Software): string {
  if (s.compliance === null && s.bank_feeds === null) {
    return "SARS filing and bank feed coverage have not been checked for this product yet, so the register makes no claim either way.";
  }
  const parts: string[] = [];
  if (s.compliance !== null) {
    parts.push(
      sarsStatus(s) === "files"
        ? "It transfers into SARS eFiling or e@syFile."
        : "It was checked and does not transfer into SARS eFiling or e@syFile.",
    );
  } else {
    parts.push("SARS filing has not been checked yet.");
  }
  if (s.bank_feeds !== null) {
    parts.push(
      s.bank_feeds.length
        ? `Bank feeds recorded: ${s.bank_feeds.join(", ")}.`
        : "No South African bank feeds are recorded.",
    );
  } else {
    parts.push("Bank feeds have not been checked yet.");
  }
  return parts.join(" ");
}

/** The assessment paragraph. Computed from the record, so it can only say what the data says. */
export function assessmentFor(s: RankedSoftware, reg: Registry, sectors: Sector[] = SECTORS): string[] {
  if (s.assessment) return s.assessment.split(/\n\n+/);
  const sector = sectorOf(s.sector, sectors);
  const peers = reg.ranked.filter((x) => x.sector === s.sector);
  const sectorRank = peers.findIndex((x) => x.slug === s.slug) + 1;

  const first =
    `${s.name} ranks ${s.rank} of ${reg.ranked.length} on the register and ${sectorRank} of ${peers.length} in ` +
    `${(sector?.label ?? "its sector").toLowerCase()}. It is rated ${formatRating(s.rating)} across ` +
    `${formatInt(s.review_count)} reviews analysed, a weighted score of ${s.score.toFixed(2)}.`;

  const p = priceLabel(s);
  let price: string;
  if (p.kind === "quoted") {
    price = "It publishes no price and quotes on request.";
  } else if (p.kind === "free") {
    price = "It starts free.";
  } else {
    const where = s.vendor_website ? ` from ${s.vendor_website}` : "";
    price = s.price_checked_at
      ? `It lists at ${randText(s)}, read${where} on ${formatDateLong(s.price_checked_at)}.`
      : `It lists at ${randText(s)}. That price has not been checked yet.`;
  }

  return [first, `${price} ${billingSentence(s)}`, complianceSentence(s)];
}

export interface Alternative {
  software: RankedSoftware;
  reason: string;
}

/** The four nearest products in the same sector by rank, each with a one line reason drawn from its own record. */
export function alternativesFor(s: RankedSoftware, reg: Registry, limit = 4): Alternative[] {
  const peers = reg.ranked.filter((x) => x.sector === s.sector && x.slug !== s.slug);
  peers.sort((a, b) => Math.abs(a.rank - s.rank) - Math.abs(b.rank - s.rank) || a.rank - b.rank);
  return peers.slice(0, limit).map((alt) => {
    const p = priceLabel(alt);
    const cost = p.kind === "free" ? "Starts free." : p.kind === "quoted" ? "Quoted on request." : `From ${randText(alt)}.`;
    const fx = p.converted ? ` Billed as ${p.original}.` : "";
    return {
      software: alt,
      reason: `Rated ${formatRating(alt.rating)} across ${formatInt(alt.review_count)} reviews. ${cost}${fx}`,
    };
  });
}

/* ------------------------------------------------------------- versus */

export interface Verdict {
  winner: RankedSoftware;
  loser: RankedSoftware;
  paragraphs: string[];
}

function sameVatFooting(a: Software, b: Software): boolean {
  const kind = (s: Software) => (s.price_basis === "mo_incl_vat" ? "incl" : s.price_basis === "mo_excl_vat" ? "excl" : "plain");
  return kind(a) === kind(b) || kind(a) === "plain" || kind(b) === "plain";
}

/**
 * The verdict names a winner and says why, using only the dataset. The winner
 * is the higher weighted score, which is the register's own ranking, so a
 * verdict can never contradict the table it sits under.
 */
export function buildVerdict(a: RankedSoftware, b: RankedSoftware): Verdict {
  const winner = a.score >= b.score ? a : b;
  const loser = winner === a ? b : a;

  let ranking =
    `${winner.name} ranks higher on the register's weighted score, ${winner.score.toFixed(2)} against ${loser.score.toFixed(2)}. ` +
    `${winner.name} is rated ${formatRating(winner.rating)} across ${formatInt(winner.review_count)} reviews analysed. ` +
    `${loser.name} is rated ${formatRating(loser.rating)} across ${formatInt(loser.review_count)}.`;
  if (loser.rating > winner.rating) {
    ranking += ` ${loser.name} has the higher raw rating, but it rests on fewer reviews, so it is pulled further toward the register average.`;
  }

  const pa = priceLabel(a);
  const pb = priceLabel(b);
  const detail: string[] = [];
  if (pa.kind === "quoted" && pb.kind === "quoted") {
    detail.push("Neither product publishes a price, so ask both for a written quote that states the VAT basis and the billing currency.");
  } else if (pa.kind === "quoted" || pb.kind === "quoted") {
    const q = pa.kind === "quoted" ? a : b;
    const o = q === a ? b : a;
    detail.push(`${q.name} publishes no price and quotes on request. ${o.name} ${priceLabel(o).kind === "free" ? "starts free" : `lists at ${randText(o)}`}.`);
  } else if (pa.kind === "free" && pb.kind === "free") {
    detail.push("Both start free.");
  } else if (!sameVatFooting(a, b)) {
    detail.push(
      `${randText(a)} for ${a.name} and ${randText(b)} for ${b.name} are not on the same VAT footing. Add ${VAT_PERCENT} percent to the figure that excludes VAT before comparing.`,
    );
  } else if (a.price_rand !== b.price_rand) {
    const cheaper = (a.price_rand ?? 0) < (b.price_rand ?? 0) ? a : b;
    const dearer = cheaper === a ? b : a;
    detail.push(`${cheaper.name} starts cheaper at ${randText(cheaper)} against ${randText(dearer)} for ${dearer.name}.`);
  } else {
    detail.push(`Both list at ${randText(a)}.`);
  }

  for (const s of [a, b]) {
    if (isForeign(s.price_currency)) {
      detail.push(`${s.name} bills in ${CURRENCY_NAME[s.price_currency!]}, so the rand figure is a conversion and its rand cost moves with the exchange rate.`);
    }
  }

  if (a.compliance === null && b.compliance === null) {
    detail.push("Neither product has been checked for SARS filing yet, so this verdict does not rest on it.");
  } else {
    const label = { files: "transfers into SARS", manual: "does not transfer into SARS", unchecked: "not yet checked for SARS" } as const;
    detail.push(`On SARS filing, ${a.name} is ${label[sarsStatus(a)]} and ${b.name} is ${label[sarsStatus(b)]}.`);
  }

  return { winner, loser, paragraphs: [ranking, detail.join(" ")] };
}

/* ---------------------------------------------------- sector standfirst */

const SECTOR_LEAD: Partial<Record<SectorSlug, string>> = {
  accounting:
    "In accounting software the two questions that cost money are whether the VAT201 transfers into SARS eFiling, and whether the product feeds from your bank.",
  payroll:
    "In payroll software the questions are EMP201, EMP501, IRP5, ETI and whether the output loads into e@syFile. A product that makes you retype any of them is a risk you pay for in penalties.",
  hr: "In HR software the question is BCEA leave accrual: whether annual, sick and family responsibility leave accrue by the rules the Act sets, and whether you can produce the record in a dispute.",
  erp: "An ERP touches VAT201 and payroll at once, and most are quote only. The questions that matter are what the quote includes and which SARS returns the product files for you.",
  crm: "CRM software holds personal information, so POPIA applies to it directly. Most of the market bills in foreign currency, which makes the exchange rate part of the monthly cost.",
  "project-management":
    "Project management software has no SARS return to file. What separates products here is price, billing currency and how quickly a free tier runs out.",
};

export function sectorStandfirst(sector: Sector, items: RankedSoftware[]): string {
  const total = items.length;
  const checked = items.filter((s) => s.compliance !== null).length;
  const priced = items.filter((s) => (s.starting_price ?? 0) > 0).length;
  const free = items.filter((s) => s.starting_price === 0).length;
  const quoted = items.filter((s) => s.starting_price === null).length;
  const foreign = items.filter((s) => isForeign(s.price_currency)).length;
  const lead = SECTOR_LEAD[sector.slug] ?? "";
  const facts =
    `The register holds ${total} ${sector.label.toLowerCase()} products: ${priced} with a published price, ${free} that start free and ${quoted} that quote only. ` +
    `${foreign} bill in a foreign currency. Compliance has been checked for ${checked} of ${total}.`;
  return `${lead} ${facts}`.trim();
}

/** Compliance items that apply to a sector, for the matrix columns. */
export function matrixItems(sector: Sector): string[] {
  return SECTOR_COMPLIANCE[sector.slug] ?? [];
}

export type MatrixCell = "yes" | "no" | "unchecked" | "na";

export function matrixCell(s: Software, item: string, applicable: string[]): MatrixCell {
  if (!applicable.includes(item)) return "na";
  if (s.compliance === null) return "unchecked";
  return s.compliance.includes(item) ? "yes" : "no";
}

/* ------------------------------------------------------------- briefings */

export function tokensFor(stats: SiteStats): Record<string, string> {
  return {
    products: String(stats.products),
    sectors: String(stats.sectors),
    comparisons: String(stats.comparisons),
    briefings: String(stats.briefings),
    reviews: formatInt(stats.reviews),
    priced: String(stats.priced),
    free: String(stats.free),
    quoted: String(stats.quoted),
    foreign: String(stats.foreign),
    zar: String(stats.zar),
    unrecorded_currency: String(stats.unrecordedCurrency),
    vat_pct: String(VAT_PERCENT),
    vat_factor: (1 + VAT_RATE).toFixed(2),
    checked_latest: stats.lastChecked ? formatDateLong(stats.lastChecked) : "not yet",
    checked_count: String(stats.checkedCount),
  };
}

export function fillTokens(text: string, tokens: Record<string, string>): string {
  return text.replace(/\{\{(\w+)\}\}/g, (_, k: string) => tokens[k] ?? `{{${k}}}`);
}

export interface VatRow {
  software: Software;
  listed: string;
  basis: string;
  including: string;
  excluding: string;
}

/** Rand products with a stated VAT basis, shown on both footings. */
export function vatRows(items: Software[]): VatRow[] {
  const rows: VatRow[] = [];
  for (const s of items) {
    if (s.price_currency !== "ZAR" || !s.starting_price) continue;
    if (s.price_basis !== "mo_incl_vat" && s.price_basis !== "mo_excl_vat") continue;
    const incl = s.price_basis === "mo_incl_vat" ? s.starting_price : roundCents(s.starting_price * (1 + VAT_RATE));
    const excl = s.price_basis === "mo_excl_vat" ? s.starting_price : roundCents(s.starting_price / (1 + VAT_RATE));
    rows.push({
      software: s,
      listed: formatMoney(s.starting_price, "ZAR"),
      basis: basisLabel(s.price_basis),
      including: formatMoney(incl, "ZAR"),
      excluding: formatMoney(excl, "ZAR"),
    });
  }
  return rows;
}

export { daysSince };
