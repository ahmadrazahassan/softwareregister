import { formatRating } from "@/lib/format";

/** A horizontal bar on a 1px --line track, filled --lime, with the numeral at the end. */
export function RatingBar({ label, value }: { label: string; value: number }) {
  const pct = Math.max(0, Math.min(100, (value / 5) * 100));
  return (
    <div className="grid grid-cols-[120px_minmax(0,1fr)_40px] items-center gap-4 py-2 text-[14px] leading-5">
      <span>{label}</span>
      <span className="block h-3 border border-line bg-surface" role="img" aria-label={`${formatRating(value)} out of 5`}>
        <span className="block h-full bg-lime" style={{ width: `${pct}%` }} />
      </span>
      <span className="text-right font-semibold">{formatRating(value)}</span>
    </div>
  );
}
