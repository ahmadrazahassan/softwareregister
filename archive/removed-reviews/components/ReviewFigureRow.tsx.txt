import Link from "next/link";
import { formatInt, formatRating } from "@/lib/format";
import type { RankedSoftware } from "@/lib/types";
import { StarGlyph } from "./primitives";
import { ProductLogo } from "./ProductLogo";

function Bar({ pct, label }: { pct: number; label: string }) {
  return (
    <span className="block h-2.5 overflow-hidden rounded-full border border-line bg-paper" role="img" aria-label={label}>
      <span className="block h-full rounded-full bg-lime" style={{ width: `${Math.max(2, Math.min(100, pct))}%` }} />
    </span>
  );
}

/**
 * One product as a row of review figures: who it is, the rating with a bar out
 * of five, the review count with a bar against the most reviewed product, and
 * the way to its reviews. A product with no logo on file shows its name only.
 */
export function ReviewFigureRow({ s, maxReviews, position }: { s: RankedSoftware; maxReviews: number; position?: number }) {
  return (
    <li className="grid gap-x-6 gap-y-3 py-4 sm:grid-cols-[minmax(0,1.3fr)_minmax(0,1fr)_minmax(0,1fr)_auto] sm:items-center">
      <div className="flex min-w-0 items-center gap-4">
        {position !== undefined && <span className="stamp w-5 flex-none text-muted">{position}</span>}
        <ProductLogo slug={s.slug} name={s.name} height={32} maxWidth={72} fallback="none" decorative className="flex-none" />
        <div className="min-w-0">
          <p className="font-heading text-[17px] leading-6 font-bold tracking-[-0.01em]">
            <Link href={`/register/${s.slug}`} className="hover:underline hover:underline-offset-4">
              {s.name}
            </Link>
          </p>
          <p className="truncate text-[12px] leading-4 text-muted">{s.vendor}</p>
        </div>
      </div>

      <div>
        <p className="mb-1.5 flex items-baseline gap-1.5">
          <span className="font-heading text-[22px] leading-6 font-bold tracking-[-0.02em]">{formatRating(s.rating)}</span>
          <StarGlyph className="text-lime-text" />
          <span className="text-[12px] leading-4 text-muted">out of 5</span>
        </p>
        <Bar pct={(s.rating / 5) * 100} label={`${formatRating(s.rating)} out of 5`} />
      </div>

      <div>
        <p className="mb-1.5 flex items-baseline gap-1.5">
          <span className="font-heading text-[22px] leading-6 font-bold tracking-[-0.02em]">{formatInt(s.review_count)}</span>
          <span className="text-[12px] leading-4 text-muted">reviews analysed</span>
        </p>
        <Bar pct={(s.review_count / maxReviews) * 100} label={`${formatInt(s.review_count)} reviews, against ${formatInt(maxReviews)} for the most reviewed product`} />
      </div>

      <Link
        href={`/register/${s.slug}/reviews`}
        className="inline-flex min-h-10 items-center justify-center rounded-[10px] border border-line-firm bg-surface px-4 text-[14px] font-semibold whitespace-nowrap hover:bg-lime-wash sm:justify-self-end"
      >
        Reviews <span aria-hidden="true" className="ml-1.5">→</span>
        <span className="sr-only"> of {s.name}</span>
      </Link>
    </li>
  );
}
